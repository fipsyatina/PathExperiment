cmake_minimum_required(VERSION 3.28)
project(PathExperiment VERSION 0.1.0 LANGUAGES CXX)

option(PATHEXPERIMENT_BUILD_APP "Build the SFML desktop application" ON)
option(PATHEXPERIMENT_BUILD_TESTS "Build headless correctness tests" ON)
option(PATHEXPERIMENT_ENABLE_GUI_TESTS "Run native smoke tests (requires a display)" OFF)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
    string(TOUPPER "${config}" CONFIG_UPPER)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER} "${CMAKE_BINARY_DIR}/bin")
endforeach()

add_library(pathx_core
    src/core/Grid.cpp
    src/core/Search.cpp
    src/core/MapStorage.cpp)
target_include_directories(pathx_core PUBLIC include)
target_compile_features(pathx_core PUBLIC cxx_std_20)

function(pathx_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
pathx_warnings(pathx_core)

if(PATHEXPERIMENT_BUILD_TESTS)
    enable_testing()
    add_executable(pathx_tests tests/core_tests.cpp)
    target_link_libraries(pathx_tests PRIVATE pathx_core)
    pathx_warnings(pathx_tests)
    add_test(NAME pathx_core_correctness COMMAND pathx_tests)
endif()

if(PATHEXPERIMENT_BUILD_APP)
    include(FetchContent)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static dependencies" FORCE)
    set(SFML_USE_STATIC_STD_LIBS ON CACHE BOOL "Portable Windows runtime" FORCE)
    set(SFML_BUILD_AUDIO OFF CACHE BOOL "Audio not used" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "Networking not used" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "No SFML examples" FORCE)
    set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "No SFML tests" FORCE)
    FetchContent_Declare(SFML
        URL https://github.com/SFML/SFML/archive/refs/tags/3.1.0.tar.gz
        URL_HASH SHA256=91209a112c2bd0bc6f4ce0d5f3e413cfb48b57c0de59f5507dc81f71b1ad7a5c
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE
        EXCLUDE_FROM_ALL SYSTEM)
    # Git's Windows checkout otherwise skips HarfBuzz's very long test-data names
    # when this project lives deep under the user's workspace. Scope the setting
    # to this configure process; do not modify the user's Git configuration.
    if(WIN32)
        set(pathx_old_git_count "$ENV{GIT_CONFIG_COUNT}")
        if(pathx_old_git_count MATCHES "^[0-9]+$")
            set(pathx_git_index "${pathx_old_git_count}")
        else()
            set(pathx_git_index 0)
        endif()
        math(EXPR pathx_git_count "${pathx_git_index} + 1")
        set(ENV{GIT_CONFIG_KEY_${pathx_git_index}} core.longpaths)
        set(ENV{GIT_CONFIG_VALUE_${pathx_git_index}} true)
        set(ENV{GIT_CONFIG_COUNT} "${pathx_git_count}")
    endif()
    FetchContent_MakeAvailable(SFML)
    if(WIN32)
        unset(ENV{GIT_CONFIG_KEY_${pathx_git_index}})
        unset(ENV{GIT_CONFIG_VALUE_${pathx_git_index}})
        if(pathx_old_git_count STREQUAL "")
            unset(ENV{GIT_CONFIG_COUNT})
        else()
            set(ENV{GIT_CONFIG_COUNT} "${pathx_old_git_count}")
        endif()
    endif()

    add_executable(PathExperiment src/app/main.cpp src/app/Application.cpp src/app/Renderer.cpp)
    target_link_libraries(PathExperiment PRIVATE pathx_core SFML::Graphics)
    pathx_warnings(PathExperiment)
    if(WIN32)
        target_link_libraries(PathExperiment PRIVATE comdlg32)
        target_link_options(PathExperiment PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup)
    endif()
    add_custom_command(TARGET PathExperiment POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:PathExperiment>/assets"
        COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/maps" "$<TARGET_FILE_DIR:PathExperiment>/maps")
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT PathExperiment)
    if(PATHEXPERIMENT_ENABLE_GUI_TESTS)
        enable_testing()
        add_test(NAME pathx_native_smoke COMMAND PathExperiment --smoke-test "${CMAKE_BINARY_DIR}/artifacts/native-smoke")
        set_tests_properties(pathx_native_smoke PROPERTIES LABELS "gui" TIMEOUT 120)
    endif()
    install(TARGETS PathExperiment RUNTIME DESTINATION .)
    install(DIRECTORY assets maps docs DESTINATION .)
    install(FILES README.md LICENSE DESTINATION .)
endif()
